\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{10pt}
  \begin{itemize}
    \item Menetapkan \textit{budget} performa berupa angka, yaitu p95 tiap alamat,
          \textit{response size}, dan ambang Core Web Vitals, lalu menegakkannya
          sebagai \textit{gate} otomatis.

    \item Menemukan bagian termahal lewat \textit{profiling}, lalu membedakan
          perbaikan yang menyentuh sebabnya dari perbaikan yang hanya
          menyentuh \textit{symptom}-nya.
  \end{itemize}

\section{Budget Performa}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{\textit{Budget} Performa}}
\end{frame}

\begin{frame}{{\Large Tiga Jenis Batas}}
  \vspace{10pt}
  \begin{itemize}
    \item \textbf{\textit{Latency}} pada persentil tertentu, biasanya p95, karena
          rata-rata menyembunyikan pengalaman terburuk.
    \item \textbf{\textit{Response size}} di kabel, karena byte yang dikirim
          menentukan waktu di jaringan yang lambat.
    \item \textbf{Core Web Vitals}, karena itulah yang benar-benar dirasakan
          pengguna di browser. Isinya tiga: LCP (\textit{Largest Contentful
          Paint}), waktu sampai elemen isi terbesar tampil; INP
          (\textit{Interaction to Next Paint}), jeda antara interaksi dan
          pembaruan layar; dan CLS (\textit{Cumulative Layout Shift}),
          banyaknya tata letak yang bergeser sendiri.
  \end{itemize}
  \vspace{6pt}
  {\footnotesize Ketiganya ditulis di satu file dan diperlakukan seperti
  kode: masuk repositori, ikut ditinjau, dan berubah lewat \textit{pull
  request}.}
\end{frame}

\begin{frame}{{\Large Analogi: Budget Belanja Kepanitiaan}}
  \vspace{14pt}
  \begin{itemize}
    \item Angkanya ditetapkan sebelum belanja dimulai, bukan sesudah nota
          menumpuk.
    \item Setiap pengeluaran baru harus muat di dalam pagu.
    \item Bila tidak muat, ada pengeluaran lain yang harus dipangkas.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Fitur baru boleh menambah waktu, asalkan masih muat, atau
  ada bagian lain yang dipercepat lebih dahulu.}
\end{frame}

function catatLcp(daftar: PerformanceObserverEntryList): void {
  const entri = daftar.getEntries();
  const terakhir = entri[entri.length - 1];
  if (terakhir !== undefined) {
    laporan.lcp_ms = Math.round(terakhir.startTime);
  }
}

function kirimLaporan(): void {
  if (sudahDikirim) {
    return;
  }
  sudahDikirim = true;
  navigator.sendBeacon(ALAMAT_LAPORAN, JSON.stringify(laporan));
}
\end{lstlisting}
  \vspace{2pt}
  {\footnotesize \texttt{startTime} diwarisi dari \texttt{PerformanceEntry},

\begin{frame}{{\Large Mengapa Dikirim Saat Halaman Ditinggalkan}}
  \vspace{14pt}
  \begin{itemize}
    \item INP baru diketahui setelah pengguna sempat berinteraksi.
    \item CLS masih dapat bertambah sampai halaman ditutup.
    \item \texttt{navigator.sendBeacon} membuat permintaannya tetap terkirim
          walaupun tabnya sudah ditutup.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Laporan yang dikirim saat halaman selesai dimuat akan selalu
  mencatat INP nol dan CLS yang belum lengkap.}
\end{frame}

\begin{frame}{{\Large Dua Aturan yang Masing-Masing Masuk Akal}}
  \vspace{10pt}
  \begin{itemize}
    \item \textbf{Algoritma Nagle} mencegah jaringan dibanjiri paket kerdil.
          Data satu byte tetap menanggung sekitar 40 byte header, jadi
          tulisan kecil ditahan sampai data sebelumnya menerima ACK, atau
          sampai terkumpul satu segmen penuh.
    \item \textbf{Delayed ACK} menekan jumlah paket tanda terima. Penerima
          menunda ACK sampai sekitar 40 milidetik, berharap dapat
          menumpangkannya pada data balasan.
  \end{itemize}
  \vspace{8pt}
  {\small Keduanya lahir di zaman jaringan lambat, dan sendiri-sendiri
  keduanya memang menghemat. Masalahnya muncul ketika bertemu.}
\end{frame}

\begin{frame}{{\Large Nagle Bertemu Delayed ACK}}
  \vspace{10pt}
  \begin{itemize}
    \item Header dan isi \textit{response} ditulis sebagai dua segmen Transmission
          Control Protocol (TCP) yang terpisah.
    \item Nagle menahan segmen kecil kedua sampai ACK segmen pertama tiba,
          sedangkan penerima sengaja menunda ACK itu. Penundaannya keputusan
          penerima, bukan akibat jaringan.
    \item Keduanya saling menunggu, dan tiap \textit{response} kecil menanggung
          tambahan tetap sekitar 40 milidetik.
  \end{itemize}
  \vspace{8pt}
  {\small Angka 40 milidetik itu timer \textit{delayed ACK} milik sistem
  operasi, bukan angka HTTP dan bukan angka aplikasi. Buktinya pengukuran ini
  berjalan lewat \textit{localhost}, yang perjalanan pulang-perginya hanya
  0,049 milidetik.}
\end{frame}

\begin{frame}{{\Large Obatnya: TCP\_NODELAY}}
  \vspace{10pt}
  \begin{itemize}
    \item \texttt{TCP\_NODELAY} adalah opsi soket yang mematikan Nagle untuk
          satu koneksi. Artinya tulisan kecil langsung dikirim tanpa menunggu
          ACK.
    \item Di Python, opsi itu dipasang dengan menyetel
          \texttt{disable\_nagle\_algorithm = True} pada kelas penangan
          permintaan.
    \item Obat lainnya bersifat struktural: menulis header dan isi sekali
          jalan, sehingga segmennya memang hanya satu.
  \end{itemize}
  \vspace{8pt}
  {\small Keduanya menyelesaikan masalah yang sama. Yang pertama satu baris,
  yang kedua menuntut penulisan ulang cara \textit{response} dikirim.}
\end{frame}

\begin{frame}{{\Large Mengirim Bagian yang Sudah Siap}}

\section{Ringkasan}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Ringkasan}}
\end{frame}

\begin{frame}{{\Large Ringkasan: Budget dan Profil}}
  \vspace{10pt}
  \begin{itemize}
    \item \textit{Budget} berisi tiga batas: \textit{latency} p95, \textit{response size}, dan ambang
          Core Web Vitals. \textit{Gate}-nya keluar dengan status 1 begitu satu
          batas terlampaui.
    \item Dua profil, dua kesimpulan: katalog penuh habis di serialisasi
          50.000 baris, agregat habis 99 persen menunggu basis data.
    \item Tempat mengukur menentukan kesimpulan. Klien mencatat 43,06
          milidetik untuk pekerjaan di bawah satu milidetik, dan sebabnya
          algoritma Nagle. Sesudah disetel, p50 menjadi 0,58 milidetik.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Ringkasan: Beban dan Waktu Tunggu}}
  \vspace{10pt}
  \begin{itemize}
    \item Alamat agregat mentok di kisaran 13 permintaan per detik. Menaikkan
          konkurensi menjadi 32 hanya menaikkan p95 dari 214,61 menjadi
          3.945,97 milidetik.
    \item \textit{Streaming} menurunkan waktu sampai byte pertama dari
          ratusan milidetik menjadi kurang dari satu milidetik, tanpa
          mengubah waktu total.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Yang membedakan perbaikan sungguhan dari perbaikan semu
  adalah bukti pengukuran sebelum dan sesudahnya.}
\end{frame}
