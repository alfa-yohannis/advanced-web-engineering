\section{Tujuan dan Kerangka Bahasan}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Tujuan dan Kerangka Bahasan}}
\end{frame}

\begin{frame}{{\Large Tujuan Pembelajaran}}
  \vspace{15pt}
  \begin{itemize}
    \item Merancang skema relasional beserta \textit{constraint}-nya, lalu menjelaskan
          akibat tiap pilihan tipe data, kunci, dan tingkat abstraksi akses
          data.
    \item Membaca rencana eksekusi PostgreSQL, lalu menentukan indeks yang
          tepat beserta kasus ketika indeks justru tidak dipakai.
    \item Membuktikan secara terukur bahwa batas transaksi, tingkat isolasi,
          dan jumlah perjalanan ke basis data menentukan latensi dan
          kebenaran data.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Hasil yang benar belum tentu murah biayanya.}
\end{frame}

\begin{frame}{{\Large Biaya yang Tidak Terlihat dari Kode}}
  \vspace{10pt}
  \begin{itemize}
    \item Pertemuan sebelumnya hanya membahasa pada bagian \emph{Waiting}, yaitu waktu sejak
          \textit{request} sampai di server hingga byte pertama ditampilkan di \textit{web browser}.
    \item Pada hampir semua aplikasi web, bagian terbesarnya dihabiskan di
          basis data.
    \item Biayanya tidak bisa terlihat dari kode:
    \begin{itemize}
      \item satu \textit{event} menjadi ratusan \textit{request} ke basis data, 
      \item satu kolom tanpa indeks tidak menunjukkan gejala perlambatan sampai tabelnya membesar,
      \item satu transaksi besar menahan \textit{request} lain.
    \end{itemize}
  \end{itemize}
  \vspace{6pt}
  {\footnotesize Ketiganya lolos pengujian fungsional karena,  hasilnya memang
  benar, tidak menguji \textit{performance}.}
\end{frame}

\section{Jenis-jenis Basis Data}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Jenis-jenis Basis Data}}
\end{frame}

\begin{frame}{{\Large Satu Fungsi, Banyak Bentuk}}
  \vspace{10pt}
  \begin{itemize}
    \item Tiap jenis basis data menyusun datanya dengan cara berbeda.
    \item Cara penyusunan itu menentukan operasi mana yang murah dan mana
          yang mahal.
    \item Memaksa satu susunan/bentuk menjawab seluruh pertanyaan selalu berakhir
          dengan \textit{side-effect} atau \textit{trade-off}.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Analoginya perpustakaan. Rak menurut nomor panggil
  memudahkan pencarian satu buku, katalog menurut subjek memudahkan
  penelusuran tema, dan daftar pinjaman harian memudahkan statistik bulanan.
  Ketiganya menyimpan keterangan tentang buku yang sama.}
\end{frame}

\begin{frame}{{\Large Cara Kerja Tiap Jenis}}
  \vspace{5pt}
  \centering
  {\footnotesize
  \begin{tabularx}{\textwidth}{lX}
    \hline
    \textbf{Jenis} & \textbf{Cara kerjanya} \\
    \hline
    Relasional & Data dipecah menjadi tabel normal, pemetaan relasi disusun saat kueri lewat penggabungan \\
    \hline
    Dokumen & Satu entitas beserta banyak atributnya disimpan utuh sebagai satu dokumen JSON \\
    \hline
    \textit{Key-Value} di memori & \textit{Value} disimpan di memori dan dicari lewat \textit{Key}-nya, tanpa pemindaian \\
    \hline
    Kolomnar & Data disimpan per kolom, bukan per baris, sehingga satu kolom dapat dibaca tanpa menyentuh kolom lain (vs relational di mana semua kolom dibaca)  \\
    \hline
    Graf & \textit{Node} dan \textit{edge} disimpan sebagai penunjuk langsung, sehingga penelusuran hubungan tidak memerlukan penggabungan \\
    \hline
    Deret waktu & Baris berstempel waktu ditulis berurutan dan di-\textit{compress} per \textit{range} \\
    \hline
    Mesin pencari & Teks dipecah menjadi kata, lalu disimpan sebagai indeks terbalik, di mana tiap kata menyimpan daftar dokumen yang memuatnya \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Cocok untuk Apa, dan Apa Biayanya}}
  \vspace{3pt}
  \centering
  {\scriptsize
  \begin{tabularx}{\textwidth}{lYY}
    \hline
    \textbf{Jenis} & \textbf{Cocok untuk} & \textbf{Trade-off} \\
    \hline
    Relasional & Data yang saling terkait dan menuntut transaksi, misalnya pesanan dan pembayaran & Skema harus dirancang lebih dahulu, dan penggabungan lintas mesin mahal \\
    \hline
    Dokumen & Isi yang selalu dibaca sekaligus dan bentuknya berbeda-beda, misalnya katalog produk & Data yang sama tersalin di banyak dokumen, sehingga pembaruannya menyebar \\
    \hline
    \textit{Key-Value} di memori & \textit{Cache}, sesi, penghitung, papan peringkat & Isi memori terbatas dan dapat hilang, sehingga bukan \textit{source of truth} \\
    \hline
    Kolomnar & Analitik atas jutaan baris, misalnya laporan dan \textit{dashboard} & Penulisan satu baris mahal, dan tidak dirancang untuk transaksi \\
    \hline
    Graf & \textit{Social network}, rekomendasi, penelusuran hak akses berjenjang & Kueri agregat biasa justru lebih lambat, dan penskalaannya rumit \\
    \hline
    Deret waktu & Metrik, sensor, dan pemantauan, sebagaimana Bab 9 & Pembaruan dan penghapusan baris lama hampir tidak didukung \\
    \hline
    Mesin pencari & Pencarian teks bebas dengan peringkat relevansi & Bukan \textit{source of truth}, dan pembaruannya tertunda beberapa saat \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Dua Contoh Pemilihan}}
  \vspace{8pt}
  \begin{enumerate}
    \item Toko buku daring menyimpan pesanan, pembayaran, dan stok di basis
          data relasional, karena ketiganya harus berubah bersama dalam satu
          transaksi. Pencarian judulnya dilayani mesin pencari, dan keranjang
          belanja sementaranya disimpan di penyimpanan \textit{key-value} di memori.
    \item \textit{Dashboard} penjualan setahun terakhir dipindahkan ke basis
          data kolomnar. Kuerinya hanya menyentuh tiga kolom dari puluhan,
          sehingga sisanya tidak perlu dibaca sama sekali.
  \end{enumerate}
  \vspace{6pt}
  {\footnotesize Tiap penyimpanan tambahan kemungkinan menambah satu \textit{source of truth},
  satu jalur penyalinan data, satu kemungkinan data tidak sinkron, dan satu
  sistem lagi yang harus dipantau serta dicadangkan.}
\end{frame}

\begin{frame}{{\Large Mengapa Relasional Menjadi Acuan}}
  \vspace{8pt}
  \begin{itemize}
    \item Jaminan transaksinya paling kuat, dan hampir seluruh aplikasi web
          memuat operasi yang menuntut jaminan itu.
    \item PostgreSQL modern sudah menyediakan kolom \texttt{jsonb}, pencarian
          teks penuh, serta ekstensi untuk data deret waktu dan vektor.
    \item Transaksi, indeks, dan rencana eksekusi punya padanan di jenis lain,
          sehingga pemahamannya dapat dipindahkan.
  \end{itemize}
  \vspace{8pt}
  {\footnotesize \textbf{Disarankan}: mulai dari basis data relasional,
  lalu tambahkan jenis lain hanya setelah pengukuran menunjukkan batasnya
  tercapai.}
\end{frame}

\section{Pemodelan Relasional dan SQL}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Pemodelan Relasional\\dan SQL}}
\end{frame}

\begin{frame}{{\Large Tabel, Kunci Primer, dan Kunci Asing}}
  \vspace{10pt}
  \begin{itemize}
    \item Model relasional menyimpan data sebagai tabel. Tiap baris mewakili
          satu fakta, tiap kolom mewakili satu sifat fakta tersebut.
    \item Baris dibedakan lewat kunci primer, sedangkan hubungan antartabel
          dinyatakan lewat kunci asing.
    \item Bahasa untuk memintanya adalah Structured Query Language (SQL).
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Analoginya arsip akademik kampus. Nomor induk mahasiswa
  dipakai sebagai penunjuk di kartu rencana studi, sehingga pergantian nama
  cukup mengubah satu baris di buku induk, bukan ratusan kartu.}
\end{frame}

\begin{frame}[fragile]{{\Large Skema Ditulis Beserta Constraint-nya}}
  \vspace{2pt}
\begin{lstlisting}[style=SqlStyle]
CREATE TABLE pesanan (
  id           bigserial PRIMARY KEY,
  pelanggan_id bigint NOT NULL REFERENCES pelanggan (id),
  status       status_pesanan NOT NULL DEFAULT 'baru',
  total        numeric(12, 2) NOT NULL DEFAULT 0,
  dibuat_pada  timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE item_pesanan (
  pesanan_id   bigint NOT NULL REFERENCES pesanan (id) ON DELETE CASCADE,
  buku_id      bigint NOT NULL REFERENCES buku (id),
  jumlah       integer NOT NULL CHECK (jumlah > 0),
  harga_satuan numeric(12, 2) NOT NULL,
  PRIMARY KEY (pesanan_id, buku_id)
);
\end{lstlisting}
\end{frame}

\begin{frame}{{\Large Constraint sebagai Penjaga Terakhir}}
  \vspace{10pt}
  \begin{itemize}
    \item Validasi di aplikasi dapat dilewati oleh skrip perbaikan data, oleh
          layanan lain yang menulis ke basis data yang sama, atau oleh versi
          aplikasi yang lupa diperbarui.
    \item \textit{Constraint} di basis data tidak dapat dilewati siapa pun.
    \item Kunci primer gabungan pada \texttt{item\_pesanan} sekaligus menjadi
          aturan bisnis: satu buku hanya boleh muncul sekali dalam satu
          pesanan.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Pilihan Tipe Data dan Akibatnya}}
  \vspace{5pt}
  \centering
  {\scriptsize
  \begin{tabularx}{\textwidth}{lX}
    \hline
    \textbf{Pilihan} & \textbf{Akibatnya} \\
    \hline
    \texttt{numeric} untuk uang & Perhitungannya tepat. \texttt{float} menyimpan pecahan biner, sehingga penjumlahan harga dapat meleset beberapa sen. \\
    \hline
    \texttt{timestamptz}, bukan \texttt{timestamp} & Waktu disimpan beserta acuan zonanya. Tanpa itu, data dari server di zona berbeda tidak dapat dibandingkan. \\
    \hline
    \texttt{text}, bukan \texttt{varchar(n)} & Di PostgreSQL keduanya sama cepat. Batas panjang lebih baik ditulis sebagai \texttt{CHECK}, karena lebih mudah diubah. \\
    \hline
    \texttt{bigint}, bukan \texttt{integer}, untuk kunci & Batas \texttt{integer} adalah 2,1 miliar. Mengubah tipe kunci pada tabel besar menuntut penulisan ulang seluruh tabel. \\
    \hline
    Tabel referensi, bukan \texttt{enum} & Menambah nilai baru pada \texttt{enum} menuntut perubahan skema, sedangkan menambah baris tidak. \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Dua Kasus Nyata}}
  \vspace{10pt}
  \begin{enumerate}
    \item Harga paket Rp 100.000 dibagi rata ke 3 item, lalu tiap bagiannya
          disimpan di kolom harga bertipe \texttt{float}. Saat laporan dibuat,
          data dibaca ulang lalu dijumlahkan. Server yang membulatkan tiap item
          ke Rp 33.333,33 lebih dahulu mendapat Rp 99.999,99. Server yang
          menjumlahkan nilai mentahnya dulu mendapat Rp 100.000,00. Laporan
          keuangan pun tidak cocok.
    \item Kolom waktu disimpan sebagai \texttt{timestamp} tanpa zona. Server
          aplikasi berjalan pada zona Asia/Jakarta, sedangkan basis data pada
          UTC. Laporan harian bergeser tujuh jam, dan pesanan yang masuk pukul
          06.00 tercatat pada tanggal sebelumnya.
  \end{enumerate}
\end{frame}

\section{ORM, Query Builder, dan Raw SQL}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{ORM, Query Builder,\\dan Raw SQL}}
\end{frame}

\begin{frame}{{\Large Tiga Tingkat Abstraksi}}
  \vspace{10pt}
  \begin{itemize}
    \item \textit{Object-Relational Mapping} (ORM) memetakan tabel menjadi
          kelas dan baris menjadi objek. SQL di-\textit{generate} seluruhnya.
    \item \textit{Query builder} menyusun SQL dari objek tabel dan kolom,
          bukan dari teks. Bentuk kuerinya masih terbaca, dan nama kolom yang
          salah ketahuan sebelum kueri dikirim.
    \item SQL langsung mengirim teks kueri apa adanya.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Analoginya berkendara. ORM transmisi otomatis, \textit{query
  builder} transmisi manual, SQL langsung mengatur sendiri tiap komponennya.
  Tidak ada yang paling benar, yang ada kecocokan dengan medannya.}
\end{frame}

\begin{frame}[fragile]{{\Large Lewat ORM}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def lewat_orm(mesin):
  """Menyusun kueri dari kelas hasil pemetaan, tanpa menulis SQL sama sekali."""
  perintah = (
      select(Pesanan.id, Pesanan.total, func.count(ItemPesanan.buku_id))
      .join(ItemPesanan, ItemPesanan.pesanan_id == Pesanan.id)
      .where(Pesanan.pelanggan_id == ID_PELANGGAN)
      .group_by(Pesanan.id)
      .order_by(Pesanan.dibuat_pada.desc())
      .limit(BATAS)
  )
  with Session(mesin) as sesi:
    return [tuple(baris) for baris in sesi.execute(perintah).all()]
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize Kebutuhannya sepuluh pesanan terakhir milik satu pelanggan
  beserta jumlah itemnya. SQL tidak ditulis sama sekali.}
\end{frame}

\begin{frame}[fragile]{{\Large Padanan SQL}}
  \vspace{2pt}
\begin{lstlisting}[style=SqlStyle, gobble=4]
    SELECT p.id, p.total, count(i.buku_id)
    FROM pesanan p
    JOIN item_pesanan i ON i.pesanan_id = p.id
    WHERE p.pelanggan_id = %s
    GROUP BY p.id
    ORDER BY p.dibuat_pada DESC
    LIMIT %s
\end{lstlisting}

  \vspace{4pt}
  \begin{itemize}
    \item Tiap metode pada kode ORM menjadi satu klausa SQL: \texttt{select}
          menjadi \texttt{SELECT}, \texttt{join} menjadi \texttt{JOIN}, dan
          seterusnya. Tabel di \texttt{FROM} disimpulkan dari kolom yang
          dipilih.
    \item Kedua \texttt{\%s} diisi \texttt{ID\_PELANGGAN} dan \texttt{BATAS}
          saat kueri dikirim.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{{\Large Lewat Query Builder}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def lewat_query_builder(mesin):
  """Menyusun SQL dari objek tabel dan kolom, tanpa memetakan tabel ke kelas."""
  pesanan = Pesanan.__table__
  item = ItemPesanan.__table__
  perintah = (
      select(pesanan.c.id, pesanan.c.total, func.count(item.c.buku_id))
      .select_from(pesanan.join(item, item.c.pesanan_id == pesanan.c.id))
      .where(pesanan.c.pelanggan_id == ID_PELANGGAN)
      .group_by(pesanan.c.id)
      .order_by(pesanan.c.dibuat_pada.desc())
      .limit(BATAS)
  )
  with mesin.connect() as koneksi:
    return [tuple(baris) for baris in koneksi.execute(perintah).all()]
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize Kolom diambil dari objek tabel lewat \texttt{.c}, misalnya
  \texttt{pesanan.c.id}, bukan dari kelas. Kueri dijalankan lewat koneksi
  biasa, tanpa \texttt{Session}.}
\end{frame}

\begin{frame}[fragile]{{\Large Lewat SQL Langsung}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def lewat_sql_langsung():
  """Mengirim SQL apa adanya lewat psycopg."""
  with psycopg.connect(DSN) as koneksi:
    baris = koneksi.execute(SQL_LANGSUNG, (ID_PELANGGAN, BATAS)).fetchall()
  return [(b[0], float(b[1]), b[2]) for b in baris]
\end{lstlisting}

  \vspace{4pt}
  \begin{itemize}
    \item Teks kueri dikirim apa adanya lewat \texttt{psycopg}.
    \item \textit{Data Source Name} (DSN) adalah satu baris teks berisi
          keterangan untuk menghubungi basis data, misalnya
          \texttt{postgresql://admin:admin123@localhost:5433/toko}.
    \item Ketiga cara menghasilkan baris yang identik, dan berkas
          \texttt{tiga-lapis-akses.py} membuktikannya setiap kali dijalankan.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Kekuatan dan Kelemahan}}
  \vspace{8pt}
  \centering
  {\small
  \begin{tabularx}{\textwidth}{lYY}
    \hline
    \textbf{Tingkat} & \textbf{Kekuatan} & \textbf{Kelemahan} \\
    \hline
    ORM & Kode singkat, struktur tabel cukup ditulis sekali di kelas model, perubahan skema dicatat sebagai berkas migrasi & SQL yang dikirim tidak terlihat, sehingga N+1 dan indeks hilang mudah lolos \\
    \hline
    Query builder & Bentuk kueri terbaca, nama kolom diperiksa lebih awal & Kueri rumit menjadi bertele-tele \\
    \hline
    SQL langsung & Yang dijalankan persis seperti yang tertulis & Kode berulang, dan perubahan skema tidak ketahuan sampai program berjalan \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Mana yang Dipilih?}}
  \vspace{8pt}
  \begin{itemize}
    \item Ketiganya dipakai dalam satu aplikasi yang sama.
    \begin{itemize}
      \item ORM untuk operasi baca dan tulis biasa yang banyak dan seragam.
      \item \textit{Query builder} untuk kueri yang bentuknya bergantung pada
            masukan pengguna, misalnya penyaringan dan pengurutan.
      \item SQL langsung untuk kueri laporan dan kueri yang perlu disetel.
    \end{itemize}
    \item Syaratnya, SQL yang benar-benar dikirim harus dapat dilihat:
          pencatatan kueri saat pengembangan, dan jumlah kueri per
          \textit{endpoint} saat pengujian.
  \end{itemize}
  \vspace{4pt}
  {\footnotesize \textit{Endpoint} adalah satu alamat yang dilayani server
  aplikasi, misalnya \texttt{GET /api/pesanan}. Jumlah kueri per
  \textit{endpoint} berarti banyaknya kueri SQL untuk melayani satu
  \textit{request} ke alamat itu.}
\end{frame}

\section{Transaksi dan Isolation Level}

\begin{frame}{{\Large Empat Sifat Transaksi}}
  \vspace{8pt}
  Transaksi adalah sekumpulan perintah yang diperlakukan sebagai satu satuan.
  Sifatnya diringkas menjadi ACID:
  \vspace{2pt}
  \begin{itemize}
    \item \textbf{\textit{Atomicity}}: seluruhnya jadi atau seluruhnya batal.
    \item \textbf{\textit{Consistency}}: \textit{constraint} basis data tetap terpenuhi
          sesudahnya.
    \item \textbf{\textit{Isolation}}: transaksi yang berjalan bersamaan tidak
          saling melihat pekerjaan setengah jadi.
    \item \textbf{\textit{Durability}}: yang sudah di-\textit{commit} tidak
          hilang walaupun listrik padam.
  \end{itemize}
  \vspace{4pt}
  {\footnotesize Analoginya transfer uang ke sesama nasabah satu bank. Bila
  saldo penerima gagal ditambah, pengurangan saldo pengirim ikut dibatalkan,
  sehingga uang tidak lenyap di tengah jalan.}
\end{frame}

\begin{frame}{{\Large Anomali Transaksi Bersamaan}}
  \vspace{5pt}
  \centering
  {\footnotesize
  \begin{tabularx}{\textwidth}{lX}
    \hline
    \textbf{Anomali} & \textbf{Kejadiannya} \\
    \hline
    \textit{Dirty read} & Membaca data yang ditulis transaksi lain yang belum di-\textit{commit}, dan ternyata dibatalkan. \\
    \hline
    \textit{Non-repeatable read} & Baris yang sama dibaca dua kali dalam satu transaksi, dan isinya sudah berubah. \\
    \hline
    \textit{Phantom read} & Kueri yang sama dijalankan dua kali, dan jumlah barisnya bertambah. \\
    \hline
    \textit{Lost update} & Dua transaksi membaca nilai yang sama, menghitung, lalu menulis. Tulisan yang terakhir menimpa yang pertama. \\
    \hline
    \textit{Write skew} & Dua transaksi membaca kondisi yang sama, lalu masing-masing menulis baris berbeda, sehingga gabungannya melanggar aturan. \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Contoh Anomali Saat Membaca}}
  \vspace{5pt}
  \centering
  {\footnotesize
  \begin{tabularx}{\textwidth}{lX}
    \hline
    \textbf{Anomali} & \textbf{Contoh} \\
    \hline
    \textit{Dirty read} & A mengubah harga buku dari Rp 80.000 menjadi Rp 60.000, tetapi belum di-\textit{commit}. B membaca Rp 60.000 dan memakainya di tagihan. A lalu dibatalkan, sehingga tagihan B memakai harga yang tidak pernah berlaku. \\
    \hline
    \textit{Non-repeatable read} & A membaca harga buku Rp 80.000. B mengubahnya menjadi Rp 90.000, lalu melakukan \textit{commit}. A membaca baris yang sama sekali lagi dan mendapat Rp 90.000. \\
    \hline
    \textit{Phantom read} & A menghitung pesanan hari ini dan mendapat 120. B menambah satu pesanan, lalu melakukan \textit{commit}. A menjalankan kueri yang sama dan mendapat 121. \\
    \hline
  \end{tabularx}}

  \vspace{8pt}
  {\footnotesize A dan B adalah dua transaksi yang berjalan bersamaan.}
\end{frame}

    \hline
  \end{tabularx}}

  {\footnotesize A dan B adalah dua transaksi yang berjalan bersamaan.}

\begin{frame}{{\Large Tingkat Isolasi pada PostgreSQL}}
  \vspace{5pt}
  \centering
  {\small
  \begin{tabularx}{\textwidth}{lYY}
    \hline
    \textbf{Tingkat} & \textbf{Masih mungkin} & \textbf{Biayanya} \\
    \hline
    Read Committed (bawaan) & Non-repeatable read, phantom read, lost update & Paling murah, hampir tidak ada transaksi yang gagal \\
    \hline
    Repeatable Read & Lost update lewat baca lalu tulis di aplikasi, write skew & Transaksi dapat gagal dengan \textit{serialization failure} dan harus diulang \\
    \hline
    Serializable & Tidak ada & Paling mahal, kegagalan yang harus diulang paling sering \\
    \hline
  \end{tabularx}}
\end{frame}

\begin{frame}{{\Large Contoh Pemakaian Tiap Tingkat}}
  \vspace{8pt}
  \begin{itemize}
    \item Read Committed: katalog, keranjang, dan pembuatan pesanan. Stok
          tetap dikurangi dengan \texttt{UPDATE} atomik atau penguncian baris,
          karena \textit{lost update} masih mungkin terjadi.
    \item Repeatable Read: laporan penjualan harian yang membaca beberapa
          tabel, agar seluruh angkanya berasal dari satu potret data.
    \item Serializable: aturan yang melibatkan banyak baris, misalnya minimal
          satu admin toko harus aktif, karena hanya tingkat ini yang mencegah
          \textit{write skew}.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Tiga Catatan dan Satu Konsekuensi}}
  \vspace{8pt}
  \begin{itemize}
    \item PostgreSQL tidak pernah membolehkan \textit{dirty read}, karena
          \textit{multiversion concurrency control} membuat pembaca melihat
          potret data yang konsisten.
    \item Repeatable Read pada PostgreSQL sudah mencegah \textit{phantom
          read}, lebih ketat daripada yang diwajibkan standar ANSI SQL.
    \item Serializable memakai \textit{serializable snapshot isolation}:
          pembaca tidak dikunci, tetapi transaksi dibatalkan bila terdeteksi
          kemungkinan pelanggaran urutan.
  \end{itemize}
  \vspace{6pt}
  {\footnotesize Begitu isolasi dinaikkan di atas Read Committed, aplikasi
  \textbf{wajib} menyediakan \textit{retry} otomatis. Tanpa itu, data yang salah
  hanya ditukar dengan \textit{error} yang sampai ke pengguna.}
\end{frame}

\begin{frame}[fragile]{{\Large Pengaman Atomik}}

\begin{lstlisting}[style=PythonStyle]
SQL_BACA_STOK_TERKUNCI = "SELECT stok FROM buku WHERE id = %s FOR UPDATE"
SQL_TULIS_STOK = "UPDATE buku SET stok = %s WHERE id = %s"
SQL_KURANGI_ATOMIK = "UPDATE buku SET stok = stok - 1 WHERE id = %s"


def atomik(koneksi):
  """Menyerahkan pembacaan dan pengurangan ke basis data sekaligus."""
  with koneksi.transaction():
    koneksi.execute(SQL_KURANGI_ATOMIK, (ID_BUKU,))
  return 1
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize Selama nilai barunya dapat dinyatakan sebagai fungsi dari
  nilai lamanya, tidak ada nilai yang singgah di aplikasi, sehingga tidak ada
  jeda yang dapat disisipi.}

\begin{frame}[fragile]{{\Large Pengaman Pesimistis}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
SQL_BACA_STOK_TERKUNCI = "SELECT stok FROM buku WHERE id = %s FOR UPDATE"
SQL_TULIS_STOK = "UPDATE buku SET stok = %s WHERE id = %s"

def pesimistis(koneksi):
  """Mengunci baris sampai transaksi selesai, sehingga worker lain menunggu."""
  with koneksi.transaction():
    baris = koneksi.execute(SQL_BACA_STOK_TERKUNCI, (ID_BUKU,)).fetchone()
    stok_terbaca = baris[0]
    koneksi.execute(SQL_TULIS_STOK, (stok_terbaca - 1, ID_BUKU))
  return 1
\end{lstlisting}

  \vspace{2pt}
  \begin{itemize}
    \item Anggapannya, \textit{conflict} pasti terjadi. \texttt{FOR UPDATE}
          membaca baris sekaligus menguncinya sampai transaksi selesai, yaitu
          saat \textit{commit} atau \textit{rollback}.
    \item \textit{Worker} lain yang juga meminta kunci baris itu berhenti di
          \texttt{SELECT} sampai kunci dilepas.
  \end{itemize}
\end{frame}

SQL_BACA_STOK_VERSI = "SELECT stok, versi FROM buku WHERE id = %s"
SQL_TULIS_BILA_VERSI_SAMA = (
    "UPDATE buku SET stok = %s, versi = versi + 1 "
    "WHERE id = %s AND versi = %s"
)
\end{lstlisting}

  \vspace{4pt}
  \begin{itemize}
    \item Anggapannya, \textit{conflict} jarang terjadi. Karena itu tidak ada
          yang dikunci.
    \item Kolom \texttt{versi} naik satu setiap kali baris ditulis.
    \item \texttt{UPDATE} hanya mengenai baris bila versinya masih sama dengan
          yang tadi dibaca. Bila versinya sudah naik, tidak ada baris yang
          berubah.
  \end{itemize}

\begin{frame}[fragile]{{\Large Pengaman Optimistis}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def optimistis(koneksi):
  """Menulis hanya bila versinya belum berubah, lalu mengulang bila kalah."""
  percobaan = 0
  while True:
    percobaan += 1
    with koneksi.transaction():
      baris = koneksi.execute(SQL_BACA_STOK_VERSI, (ID_BUKU,)).fetchone()
      stok_terbaca, versi_terbaca = baris
      hasil = koneksi.execute(
          SQL_TULIS_BILA_VERSI_SAMA,
          (stok_terbaca - 1, ID_BUKU, versi_terbaca),
      )
    # rowcount nol berarti versinya sudah diubah worker lain.
    if hasil.rowcount == 1:
      return percobaan
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize \texttt{rowcount} nol berarti \textit{worker} lain sudah
  lebih dahulu menulis. Perulangan lalu membaca ulang dan mencoba lagi.}
\end{frame}

\begin{frame}[fragile]{{\Large Pola N+1 di Kode}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle]
def ambil_dengan_n_plus_1(koneksi, id_pelanggan, batas):
  """Mengambil daftar pesanan, lalu itemnya satu kueri untuk tiap pesanan."""
  with koneksi.cursor() as kursor:
    kursor.execute(SQL_DAFTAR_PESANAN, (id_pelanggan, batas))
    daftar_pesanan = kursor.fetchall()
    jumlah_kueri = 1

    hasil = []
    for id_pesanan, total, waktu in daftar_pesanan:
      kursor.execute(SQL_ITEM_SATU_PESANAN, (id_pesanan,))
      hasil.append((id_pesanan, total, waktu, kursor.fetchall()))
      jumlah_kueri += 1

  return hasil, jumlah_kueri
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize Kueri kedua ada di dalam perulangan, sehingga dijalankan
  sekali untuk tiap pesanan: 25 pesanan berarti 1 + 25 = 26 kueri.}
\end{frame}

\begin{frame}[fragile]{{\Large Satu per Satu atau Sekaligus}}
  \vspace{2pt}
\begin{lstlisting}[style=PythonStyle, basicstyle=\ttfamily\fontsize{7}{8.5}\selectfont]
SQL_ITEM_SATU_PESANAN = """
    SELECT b.judul, i.jumlah FROM item_pesanan i
    JOIN buku b ON b.id = i.buku_id
    WHERE i.pesanan_id = %s
"""

SQL_ITEM_BANYAK_PESANAN = """
    SELECT i.pesanan_id, b.judul, i.jumlah FROM item_pesanan i
    JOIN buku b ON b.id = i.buku_id
    WHERE i.pesanan_id = ANY(%s)
"""
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize
  \begin{itemize}
    \item Kueri pertama mengambil item satu pesanan, sehingga harus dijalankan
          sekali untuk tiap pesanan. Inilah sumber N+1.
    \item Kueri kedua menerima seluruh id pesanan sebagai larik lewat
          \texttt{ANY(\%s)}, sehingga cukup dijalankan sekali.
  \end{itemize}}
\end{frame}

def ambil_dengan_dua_kueri(koneksi, id_pelanggan, batas):
  """Mengambil data yang sama dengan jumlah kueri yang tidak bergantung N."""
  with koneksi.cursor() as kursor:
    kursor.execute(SQL_DAFTAR_PESANAN, (id_pelanggan, batas))
    daftar_pesanan = kursor.fetchall()

    semua_id = [pesanan[0] for pesanan in daftar_pesanan]
    kursor.execute(SQL_ITEM_BANYAK_PESANAN, (semua_id,))

    item_per_pesanan = defaultdict(list)
    for id_pesanan, judul, jumlah in kursor.fetchall():
      item_per_pesanan[id_pesanan].append((judul, jumlah))
\end{lstlisting}

  \vspace{2pt}
  {\footnotesize Pada 25 pesanan, 26 kueri dalam 7,28 milidetik menjadi 2
  kueri dalam 1,07 milidetik untuk 75 item yang sama persis. Satu kueri
  \texttt{JOIN} juga bisa, tetapi kolom pesanan terulang di setiap baris
  item.}
\end{frame}

\begin{frame}{{\Large Angka Lokal Meremehkan Masalah}}
  \vspace{8pt}
  \begin{itemize}
    \item Pengukuran tadi memakai basis data di komputer yang sama, sehingga
          satu perjalanan hampir tidak berbiaya.
    \item Di produksi, aplikasi dan basis data berada di mesin berbeda dengan
          RTT sekitar 1 milidetik.
    \item 26 kueri menanggung tambahan 26 milidetik, 2 kueri hanya 2
          milidetik. Pada 100 baris, selisihnya 100 milidetik lawan 2
          milidetik.
    \item Seluruhnya masuk ke bagian \emph{Waiting} yang diukur pada Bab 2.
  \end{itemize}
  \vspace{6pt}
  {\footnotesize Cara menemukannya bukan membaca kode satu per satu, melainkan
  menghitung jumlah kueri per \textit{endpoint} saat uji integrasi, lalu
  menegakkan ambang batasnya di \textit{Continuous Integration}.}
\end{frame}

\section{Indexing dan Query Plan}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Indexing dan\\Pembacaan Query Plan}}
\end{frame}

\begin{frame}{{\Large Indeks dan Biayanya}}
  \vspace{10pt}
  \begin{itemize}
    \item Indeks adalah struktur data tambahan yang memungkinkan basis data
          menemukan baris tanpa membaca seluruh tabel.
    \item Bentuk bawaannya B-tree, yaitu pohon terurut dan berimbang.
          Pencarian turun dari akar ke daun, sehingga tabel \texttt{pesanan}
          yang berisi 200.000 baris cukup ditelusuri beberapa tingkat saja.
          Karena isinya terurut, indeks yang sama juga melayani \textit{range}
          dan \texttt{ORDER BY}.
    \item Tiap indeks memperlambat penulisan dan memakan ruang.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Analoginya indeks di belakang buku. Halaman yang dituju
  langsung ketahuan, tetapi indeks harus disusun ulang setiap kali isi buku
  berubah, dan indeks itu sendiri memakan halaman.}
\end{frame}

\begin{frame}{{\Large Jenis Indeks}}
  \vspace{5pt}
  {\centering
  {\scriptsize
  \begin{tabularx}{\textwidth}{lYY}
    \hline
    \textbf{Jenis} & \textbf{Dipakai untuk} & \textbf{Contoh} \\
    \hline
    B-tree kolom tunggal & Kesamaan, \textit{range}, dan pengurutan pada satu kolom. Bawaan PostgreSQL. & \texttt{item\_pesanan (buku\_id)} \\
    \hline
    B-tree gabungan & Penyaringan pada beberapa kolom sekaligus. Hanya terpakai bila kolom terdepan ikut disaring. & \texttt{pesanan (pelanggan\_id, dibuat\_pada DESC)} \\
    \hline
    Indeks parsial & Sebagian baris saja, lewat klausa \texttt{WHERE}. Lebih kecil dan lebih cepat. & \texttt{pesanan (dibuat\_pada) WHERE status <> 'batal'} \\
    \hline
    Indeks ekspresi & Penyaringan atas hasil fungsi. & \texttt{pelanggan (lower(email))} \\
    \hline
    GIN dengan trigram & Pencarian pola di tengah teks, yang tidak dapat dilayani B-tree. & \texttt{buku USING gin (judul gin\_trgm\_ops)} \\
    \hline
  \end{tabularx}}\par}

  \vspace{6pt}
  {\footnotesize Indeks \texttt{(pelanggan\_id, dibuat\_pada)} melayani
  penyaringan atas \texttt{pelanggan\_id} saja atau keduanya, tetapi tidak atas
  \texttt{dibuat\_pada} saja.}
\end{frame}

\begin{frame}{{\Large Membaca Rencana Eksekusi}}
  \vspace{4pt}
  \begin{itemize}
    \item \texttt{EXPLAIN} memperlihatkan rencana, sedangkan
          \texttt{EXPLAIN ANALYZE} menjalankannya dan melaporkan waktu
          nyatanya.
    \item Rencananya berbentuk pohon. Tiap \textit{node} adalah satu langkah
          kerja, dan baris yang paling menjorok ke dalam dikerjakan lebih
          dahulu.
    \item Tiga hal yang paling penting dibaca:
    \begin{itemize}
      \item selisih \texttt{rows} perkiraan dan \texttt{rows} sebenarnya,
            karena selisih besar berarti statistiknya usang;
      \item \texttt{Rows Removed by Filter}, karena angka besar adalah tanda
            indeks yang hilang;
      \item \texttt{loops}, karena waktu yang tertulis adalah waktu per satu
            \textit{loop}, bukan totalnya.
    \end{itemize}
  \end{itemize}

\end{frame}

\begin{frame}[fragile]{{\Large Contoh Membaca Rencana}}

        ->  Parallel Seq Scan on pesanan  (actual time=1.175..3.531 rows=3 (*\textcolor{red}{loops=3}*))

\end{lstlisting}

  \vspace{6pt}
  \begin{itemize}
    \item Penyaring membuang 66.663 baris hanya untuk memperoleh 10 baris,

    \item Waktu 3,531 milidetik itu berlaku per satu \textit{loop}, padahal
          \textit{loop}-nya ada tiga.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Node untuk Membaca Tabel}}

\section{Connection Pooling}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Connection Pooling}}
\end{frame}

\begin{frame}{{\Large Membuka Koneksi Itu Mahal}}
  \vspace{10pt}
  \begin{itemize}
    \item Tiap koneksi PostgreSQL dilayani satu proses tersendiri di sisi
          server, lengkap dengan memorinya.
    \item Membuka koneksi berarti membangun soket, melakukan
          \textit{handshake} autentikasi, dan menyiapkan proses baru.
    \item Biayanya jauh lebih besar daripada menjalankan satu kueri
          sederhana.
  \end{itemize}
  \vspace{10pt}
  {\footnotesize Analoginya loket layanan kemahasiswaan. Membuka loket baru
  menuntut petugas, meja, dan komputer sendiri. \textit{Connection pool} adalah
  sejumlah loket yang dibuka sekali di awal, lalu dipakai bergantian. Satu
  loket melayani satu mahasiswa pada satu waktu.}
\end{frame}

\begin{frame}{{\Large Mengapa Pool Sedikit Lebih Lambat}}
  \vspace{8pt}
  \begin{itemize}
    \item \textit{Pool} menambah pekerjaan meminjam dan mengembalikan koneksi,
          lengkap dengan penguncian antarutas.
    \item Biaya itu terbayar begitu jumlah \textit{worker} melampaui jumlah koneksi
          yang boleh dibuka.
    \item 40 \textit{worker} tanpa \textit{pool} menabrak \texttt{max\_connections} yang
          disetel 25, dan 10 di antaranya gagal dengan \textit{error} \texttt{too many
          clients already}.
    \item 40 \textit{worker} yang sama lewat \textit{pool} berisi 10 koneksi seluruhnya
          berhasil dalam 43,9 milidetik.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Dua Kaidah Penyetelan}}
  \vspace{8pt}
  \begin{enumerate}
    \item Ukuran \textit{pool} ditentukan kemampuan basis data, bukan jumlah \textit{worker}
          aplikasi. Jumlah seluruh \textit{pool} dari seluruh instans harus lebih
          kecil daripada \texttt{max\_connections}, dengan sisa untuk
          administrasi.
    \item \textit{Pool} yang terlalu besar tidak mempercepat apa pun. Setelah
          prosesor dan \textit{disk} jenuh, koneksi tambahan hanya memperpanjang
          antrean.
  \end{enumerate}
\end{frame}

\begin{frame}{{\Large Serverless: Pool per Instans}}
  \vspace{8pt}
  \begin{itemize}
    \item Server biasa menyala terus. Seluruh \textit{request} meminjam
          koneksi dari satu \textit{pool} yang sama, misalnya 10 koneksi.
    \item Di \textit{serverless} tidak ada server yang menyala terus.
          Tiap \textit{request} yang datang bersamaan dilayani salinan
          aplikasi terpisah, disebut instans.
    \item Pada Amazon Web Services (AWS) Lambda, 100 \textit{request}
          bersamaan dilayani 100 instans.
    \item Tiap instans membuka \textit{pool}-nya sendiri, dan \textit{pool}
          itu tidak dapat dipinjamkan ke instans lain.
  \end{itemize}
  \vspace{6pt}
  {\footnotesize Analoginya, setiap mahasiswa yang datang minta dibukakan
  loketnya sendiri. Seratus mahasiswa berarti seratus loket, padahal
  ruangannya terbatas.}
\end{frame}

\begin{frame}{{\Large PgBouncer dan Modenya}}
  \vspace{5pt}
  {\centering
  {\small
  \begin{tabularx}{\textwidth}{lYY}
    \hline
    \textbf{Mode} & \textbf{Koneksi dipinjamkan selama} & \textbf{Akibatnya} \\
    \hline
    \texttt{session} & Klien masih terhubung & Paling aman, tetapi satu klien yang terhubung menahan satu koneksi server \\

  langsung dipakai instans lain. Karena transaksi hanya beberapa milidetik,

\begin{frame}{{\Large Jebakan Mode Transaction}}
  \vspace{5pt}
  {\centering
  {\footnotesize
  \begin{tabularx}{\textwidth}{lYY}

\end{frame}

\section{Migrasi Berversi}

\begin{frame}{\hfill}
  \centering
  \Huge{\textbf{Migrasi sebagai\\Artefak Berversi}}
\end{frame}

\begin{frame}{{\Large Tiga Aturan Migrasi}}
  \vspace{8pt}
  Perubahan skema ditulis sebagai berkas bernomor di repositori, diperiksa
  lewat \textit{code review}, dan dijalankan oleh \textit{pipeline} yang sama
  untuk seluruh lingkungan.
  \vspace{2pt}
  \begin{enumerate}
    \item Migrasi yang sudah dijalankan tidak pernah diubah lagi. Perubahan
          berikutnya ditulis sebagai berkas baru.
    \item Tiap migrasi dapat dijalankan pada basis data kosong maupun yang
          sudah berisi.
    \item Migrasi dijalankan sebelum versi aplikasi baru dinyalakan, sehingga
          versi lama dan skema baru sempat berjalan bersamaan.
  \end{enumerate}
\end{frame}

\begin{frame}{{\Large Pola Expand-Contract}}
  \vspace{10pt}
  {\centering
  \scalebox{1.15}{
  \begin{tikzpicture}[

    \node[tahap, fill=orange!12, right=of b] (c)

    \node[tahap, fill=green!10, right=of c] (d)

  \vspace{10pt}
  {\footnotesize Perubahan yang merusak, misalnya mengganti nama sebuah
  kolom, dipecah menjadi beberapa langkah. Tiap langkah aman dijalankan
  sementara versi lama aplikasi masih berjalan.}
\end{frame}

\begin{frame}{{\Large Perintah yang Mengunci Tabel}}
  \vspace{5pt}
  {\centering
  {\footnotesize
  \begin{tabularx}{\textwidth}{lX}
    \hline
    \textbf{Perintah} & \textbf{Risikonya} \\
    \hline
    \texttt{ADD COLUMN} tanpa \texttt{NOT NULL} & Aman. Kuncinya hanya sesaat. \\
    \hline
    \texttt{ADD COLUMN} dengan nilai bawaan & Aman sejak PostgreSQL 11, karena tabelnya tidak ditulis ulang. \\
    \hline
    \texttt{CREATE INDEX} & Berbahaya. Seluruh penulisan tertahan sampai indeks selesai dibangun. \\
    \hline
    \texttt{CREATE INDEX CONCURRENTLY} & Aman, tetapi lebih lama, tidak boleh di dalam transaksi, dan dapat menyisakan indeks tak sah bila gagal. \\
    \hline
    \texttt{ALTER COLUMN TYPE} & Berbahaya. Umumnya menulis ulang seluruh tabel. \\
    \hline
    \texttt{ADD CONSTRAINT} langsung & Berbahaya. Seluruh baris diperiksa sambil tabel terkunci. \\
    \hline
  \end{tabularx}}\par}

  \vspace{8pt}
  {\footnotesize Pada tabel berisi puluhan juta baris, kunci yang
  menghentikan pembacaan dan penulisan berarti layanan berhenti selama
  beberapa menit.}
\end{frame}

\begin{frame}[fragile]{{\Large Migrasi Indeks yang Aman}}
  \vspace{2pt}
\begin{lstlisting}[style=SqlStyle]
-- Tanpa BEGIN. CREATE INDEX CONCURRENTLY tidak boleh di dalam transaksi.
SET lock_timeout = '5s';

CREATE INDEX CONCURRENTLY IF NOT EXISTS idx_pesanan_pelanggan_waktu
  ON pesanan (pelanggan_id, dibuat_pada DESC);

-- Memeriksa indeks yang tertinggal dalam keadaan tidak sah:
-- SELECT indexrelid::regclass FROM pg_index WHERE NOT indisvalid;
\end{lstlisting}

  \vspace{2pt}
  \begin{itemize}
    \item Ditulis tanpa \texttt{BEGIN}, karena \texttt{CREATE INDEX
          CONCURRENTLY} tidak boleh berada di dalam blok transaksi.
    \item \texttt{lock\_timeout} membuat migrasi menyerah lebih dahulu,
          alih-alih mengantre di belakang transaksi panjang sambil menahan
          seluruh \textit{request} sesudahnya.
  \end{itemize}
\end{frame}

\begin{frame}{{\Large Ringkasan}}
  \vspace{8pt}
  \begin{itemize}
    \item \textit{Constraint} di dalam skema adalah penjaga terakhir, dan tipe data yang
          keliru baru terasa ketika laporan tidak cocok atau tanggal bergeser.
    \item ORM, \textit{query builder}, dan SQL langsung dipakai bersama,
          asalkan SQL yang benar-benar dikirim tetap terlihat.
    \item Batas transaksi dibuat sempit. Pola baca lalu tulis tanpa pengaman
          kehilangan 350 dari 400 pembaruan.
    \item Indeks dipilih dari rencana eksekusi, bukan dari kebiasaan, dan
          manfaatnya bergantung pada selektivitas penyaring.
    \item Koneksi dipinjam dari \textit{pool}, dan perubahan skema ditulis sebagai
          migrasi berversi.
    \item Yang dibaca bukan lagi panel Network, melainkan rencana eksekusi.
  \end{itemize}
\end{frame}
